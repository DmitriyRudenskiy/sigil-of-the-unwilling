function initRun(){
  villagers=[]; vid=0; repScore=0; season=0;
  stormActive=false; mourning=0;
  ACTIVE.length=0;
  const keys=Object.keys(SPECIES).sort(()=>Math.random()-.5);
  ACTIVE.push(...keys.slice(0,3));
  sStat={};

}

let HOUS={total:0,gen:0,frog:0,spBeds:{},ratio:{},shelterOk:true,frogCnt:0,nonFrog:0};
function recomputeHousing(){
  let total=0, frog=0;
  const spBeds={}; ACTIVE.forEach(k=>spBeds[k]=0);
  buildings.forEach(b=>{
    const D=B[b.type]; if(!D.housing) return;
    total+=D.housing;
    if(b.houseSp){ spBeds[b.houseSp]+=D.housing; if(b.houseSp==='Frog') frog+=D.housing; }
  });
  HOUS.total=total; HOUS.frog=frog; HOUS.gen=total-frog; HOUS.spBeds=spBeds;
}
const spCount = sp => villagers.filter(v=>v.species===sp).length;
function bedsFree(sp){
  if(sp==='Frog') return HOUS.frog - spCount('Frog');
  return HOUS.gen - (villagers.length - spCount('Frog'));
}
function housingBonus(sp){
  const r=HOUS.ratio[sp]; if(r===undefined) return 0;
  if(r>=1) return 8;
  if(r>0) return Math.round(8*r);
  return sp==='Frog' ? -12 : -6;
}
function pushVillager(){
  const cand=ACTIVE.filter(sp=>bedsFree(sp)>0);
  if(!cand.length) return false;
  const sp=cand[(Math.random()*cand.length)|0];
  villagers.push({id:++vid, species:sp, job:null, resolve:0});
  return true;
}
function fillHousing(){
  recomputeHousing();
  let added=0;
  for(let guard=0;guard<999;guard++){ if(!pushVillager()) break; added++; }
  return added;
}
function loseVillager(v){
  const i=villagers.indexOf(v); if(i<0) return 0;
  villagers.splice(i,1);
  if(v.species==='Bat') mourning=0; else mourning++;
  return 1;
}
function trimToHousing(){
  recomputeHousing();
  let removed=0;
  while(spCount('Frog')>HOUS.frog){
    let i=villagers.findIndex(v=>v.species==='Frog'&&!v.job);
    if(i<0) i=villagers.findIndex(v=>v.species==='Frog');
    if(i<0) break;
    removed+=loseVillager(villagers[i]);
  }
  while(villagers.length>HOUS.total){
    let i=villagers.findIndex(v=>!v.job);
    if(i<0) i=0;
    removed+=loseVillager(villagers[i]);
  }
  return removed;
}
const villagersIn = b => villagers.filter(v=>v.job===b);
const freeVillagers = () => villagers.filter(v=>!v.job);

let NEED={cap:{},dem:{},ok:{}};
function recomputeNeeds(){
  const cap={food:0,cloth:0,Leisure:0,Religion:0,Education:0,Treatment:0,Brawling:0,Luxury:0};
  buildings.forEach(b=>{
    const D=B[b.type]; if(!D.prov) return;
    const ratio = D.jobs ? Math.min(1,(b.workers||0)/D.jobs) : 1;
    for(const k in D.prov) cap[k]+=Math.ceil(D.prov[k]*ratio);
  });
  const dem={food:villagers.length, cloth:villagers.length};
  villagers.forEach(v=> SPECIES[v.species].services.forEach(s=> dem[s]=(dem[s]||0)+1));
  const ok={};
  for(const k in cap) ok[k]= dem[k]===0 ? true : cap[k]>=dem[k];
  NEED={cap,dem,ok};
}

function recomputeResolve(){
  const frogCnt=spCount('Frog'), nonFrog=villagers.length-frogCnt;
  HOUS.frogCnt=frogCnt; HOUS.nonFrog=nonFrog;
  HOUS.shelterOk = HOUS.gen >= nonFrog;
  ACTIVE.forEach(k=>{
    const cap = k==='Frog'? HOUS.frog : (HOUS.spBeds[k]||0);
    const cnt = spCount(k);
    HOUS.ratio[k] = cnt? Math.min(1,cap/cnt) : 1;
  });
  const humanPerk  = villagers.some(v=>v.job&&v.species==='Human');
  const lizardPerk = villagers.some(v=>v.job&&v.species==='Lizard');
  const batBonus   = Math.min(6, Math.floor(mourning/2));
  villagers.forEach(v=>{
    const S=SPECIES[v.species];
    let r=S.base + (NEED.ok.food?5:-10) + (NEED.ok.cloth?3:-8);
    S.services.forEach(s=> r += NEED.ok[s]?3:-6);
    if(v.species==='Frog'){
      r += housingBonus('Frog');
    } else {
      r += HOUS.shelterOk?4:-10;
      r += housingBonus(v.species);
    }
    if(stormActive){
      const pen = STORM_PEN[S.res] * (humanPerk?0.75:1);
      r -= Math.round(pen*10)/10;
    }
    if(lizardPerk) r += 1;
    if(v.species==='Bat') r += batBonus;
    if(v.job){
      const spec=B[v.job.type].spec;
      if(spec===S.comfort) r+=5;
      if(S.comfort==='Cooperation'&&v.species==='Fox'){
        const foxes=villagersIn(v.job).filter(w=>w.species==='Fox').length;
        if(foxes>=3) r+=5;
      }
    }
    v.resolve=r;
  });
  ACTIVE.forEach(k=>{
    const vs=villagers.filter(v=>v.species===k), st=sStat[k];
    st.count=vs.length;
    st.avg=vs.length? vs.reduce((s,v)=>s+v.resolve,0)/vs.length : 0;
    st.workers=vs.filter(v=>v.job).length;
  });
}

let autoOn=true;
function nextSeason(){
  season++;
  stormActive = (season%3===0);
  recalcAll();
  pulse({q:0,r:0}, stormActive?.9:.45);
  let gained=0, left=0;
  ACTIVE.forEach(k=>{
    const S=SPECIES[k], st=sStat[k];
    if(st.count>0 && st.avg>=st.demandCur){
      st.rep++; st.demandCur+=S.decadence; repScore+=15; gained++;
      spawnFloat(0,-R*1.5,`+слава · ${S.name}`,'pos');
    }
  });
  const leavers=villagers.filter(v=>v.resolve<1);
  leavers.forEach(v=>loseVillager(v));
  left=leavers.length;
  if(left) spawnFloat(0,-R*.2,`−${left} ушло`,'neg');
  if(season%2===0){
    const frogPerk=villagers.some(v=>v.job&&v.species==='Frog');
    let n=0;
    for(let i=0;i<2+(frogPerk?1:0);i++){ if(pushVillager()) n++; }
    if(n) spawnFloat(R*2,-R*1.5,`+караван ×${n}`,'pos');
  }
  recalcAll();
  status(`Сезон ${season}${stormActive?' · ШТОРМ':''}: слава +${gained*15}${left?` · ушло жителей: ${left} (покой ниже 1)`:''}`);
}

function genMap(){
  [...buildings].forEach(b=>removeBuilding(b,true));
  initRun();
  pin=null; pinSp=null; gPin.innerHTML='';
  const seeds=[];
  const bag=['Rural','Rural','Urban','Urban','Thicket','Swamp','Stone','Arid'];
  for(const b of bag){
    let c,tries=0;
    do{ c=cells[(Math.random()*cells.length)|0]; tries++; }
    while(tries<60 && seeds.some(s=>hexDist(s.c.q,s.c.r,c.q,c.r)<2));
    seeds.push({b,c});
  }
  cells.forEach(c=>{
    const sorted=seeds.map(s=>({s,d:hexDist(s.c.q,s.c.r,c.q,c.r)+Math.random()*.6})).sort((a,b)=>a.d-b.d);
    c.biome=(Math.random()<.12&&sorted[1])?sorted[1].s.b:sorted[0].s.b;
    c.feature=null;
  });
  const edge=cells.filter(c=>c.ring===RINGS);
  let cur=edge[(Math.random()*edge.length)|0];
  let dir=DIRS[(Math.random()*6)|0];
  for(let i=0;i<8&&cur;i++){
    cur.feature='river';
    const want=Math.random()<.7?dir:DIRS[(Math.random()*6)|0];
    let nxt=cellMap.get(key(cur.q+want[0],cur.r+want[1]));
    if(!nxt||nxt.feature==='river'){
      for(const d of DIRS.slice().sort(()=>Math.random()-.5)){
        const t=cellMap.get(key(cur.q+d[0],cur.r+d[1]));
        if(t&&!t.feature){ nxt=t; dir=d; break; }
      }
    } else dir=want;
    cur=nxt;
  }
  const free=cells.filter(c=>!c.feature&&c.ring>=1);
  const ruins=free[(Math.random()*free.length)|0];
  if(ruins) ruins.feature='ruins';
  const far=cells.filter(c=>!c.feature&&c.ring>=2&&(!ruins||hexDist(c.q,c.r,ruins.q,ruins.r)>=3));
  const pred=far[(Math.random()*far.length)|0];
  if(pred) pred.feature='predator';
}

function tryAutoMerge(cell){
  const start=cell.b;
  if(!start||start.big||start.cells.length!==1) return null;
  const t=start.type;
  /* BFS по связной группе однотипных одноклеточных зданий */
  const order=[start]; const seenB=new Set([start.id]);
  const seenC=new Set([key(cell.q,cell.r)]);
  const queue=[cell];
  while(queue.length){
    const c=queue.shift();
    for(const d of DIRS){
      const nk=key(c.q+d[0],c.r+d[1]);
      if(seenC.has(nk)) continue;
      seenC.add(nk);
      const n=cellMap.get(nk); if(!n) continue;
      const nb=n.b;
      if(nb&&!nb.big&&nb.cells.length===1&&nb.type===t){
        if(!seenB.has(nb.id)){ seenB.add(nb.id); order.push(nb); }
        queue.push(n);
      }
    }
  }
  if(order.length<4) return null;
  const four=order.slice(0,4);
  const cellsU=four.map(b=>b.cells[0]);
  const carried=four.flatMap(bl=>villagersIn(bl));
  const sps=four.map(b=>b.houseSp);
  const houseSp=sps.every(s=>s===sps[0])?sps[0]:null;
  const hadPin=four.some(b=>b===pin);
  four.forEach(b=>removeBuilding(b,true));
  const bk=ensureBigType(t);
  const nb=placeBuilding(bk,cellsU,null,{big:true,baseType:t,houseSp});
  carried.slice(0,B[bk].jobs).forEach(v=>v.job=nb);
  if(hadPin){ pin=nb; highlightPin(); showInfoBuilding(nb); }
  pulse(nb.anchor,1.6);
  spawnFloat(nb.cx,nb.cy-R*1.05,B[bk].name,'name');
  spawnFloat(nb.cx,nb.cy+R*.35,'слияние ×4','pos');
  return nb;
}

function matchScore(v,b){
  const S=SPECIES[v.species], spec=B[b.type].spec;
  if(!spec) return 0;
  let m=0; if(S.prof===spec)m+=2; if(S.comfort===spec)m+=1; return m;
}
function assignWorkers(b,d){
  const D=B[b.type], cur=villagersIn(b).length;
  if(d>0){
    if(cur>=D.jobs){ status('Рабочие места заняты'); return; }
    const free=freeVillagers();
    if(!free.length){ status('Нет свободных жителей — постройте жильё'); return; }
    free.sort((a,c)=>matchScore(c,b)-matchScore(a,b));
    free[0].job=b;
  } else {
    const ws=villagersIn(b); if(!ws.length) return;
    ws[ws.length-1].job=null;
  }
  recalcAll();
  if(pin===b) showInfoBuilding(b);
  status(`${D.name}: ${villagersIn(b).length}/${D.jobs} · свободных жителей: ${freeVillagers().length}`);
}

function minDist(a,b){
  let m=1e9;
  for(const ca of a.cells)for(const cb of b.cells) m=Math.min(m,hexDist(ca.q,ca.r,cb.q,cb.r));
  return m;
}
function minDistCell(cell,b){
  let m=1e9;
  for(const cb of b.cells) m=Math.min(m,hexDist(cell.q,cell.r,cb.q,cb.r));
  return m;
}
function aggAdj(source,D,radiusFn){
  const per=new Map();
  for(const o of buildings){
    if(source&&o===source) continue;
    if(radiusFn(o)<=D.radius){
      const v=(D.adj||{})[typeOfB(o)];
      if(v){ const e=per.get(typeOfB(o))||{v:0,n:0}; e.v+=v; e.n++; per.set(typeOfB(o),e); }
    }
  }
  const parts=[];
  for(const [t,e] of per) parts.push([B[t].name+(e.n>1?' ×'+e.n:''),e.v]);
  parts.sort((a,b)=>b[1]-a[1]);
  return parts;
}
function computeScore(b){
  const D=B[b.type], c=b.anchor, ws=villagersIn(b);
  const parts=[['база',D.base]];
  if(D.biome&&D.biome[c.biome]) parts.push(['биом · '+BIOMES[c.biome].name,D.biome[c.biome]]);
  if(D.feat&&c.feature&&D.feat[c.feature]) parts.push([FEATN[c.feature],D.feat[c.feature]]);
  if(ws.length&&D.perWorker) parts.push([`${(D.jobName||'работники').toLowerCase()} ×${ws.length}`,ws.length*D.perWorker]);
  if(D.spec){
    const prof=ws.filter(v=>SPECIES[v.species].prof===D.spec).length;
    if(prof) parts.push([`сноровка · ${SPECS[D.spec].name} ×${prof}`,prof]);
  }
  parts.push(...aggAdj(b,D,o=>minDist(b,o)));
  return {total:parts.reduce((s,p)=>s+p[1],0),parts};
}

}

let OFFERS=[];
function combos(lists,i=0,acc=[]){
  if(i===lists.length) return [acc];
  const out=[];
  for(const x of lists[i]) out.push(...combos(lists,i+1,[...acc,x]));
  return out;
}
function findOffers(){
  const out=[];
  for(const rec of RECIPES){
    const lists=rec.parts.map(t=>[...buildings].filter(b=>b.type===t));
    if(lists.some(l=>!l.length)) continue;
    for(const inst of combos(lists)){
      let ok=true;
      for(let i=0;i<inst.length&&ok;i++)
        for(let j=i+1;j<inst.length&&ok;j++)
          if(minDist(inst[i],inst[j])!==1) ok=false;
      if(ok) out.push({rec,inst});
    }
  }
  return out;
}

  Object.entries(D.adj||{}).sort((a,b)=>b[1]-a[1]).forEach(([k,v])=>hh+=row('сосед · '+(B[k]?B[k].name:k),v));
  hh+=`<div class="row"><span>слияние</span><span>${(D.merges||[]).map(m=>B[m].name).join(' · ')||'—'}</span></div>`;
  fcard.innerHTML=hh; showFloat();
}

function computeGhost(c,t){
  const D=B[t];
  const parts=[['база',D.base]];
  if(D.biome&&D.biome[c.biome]) parts.push(['биом · '+BIOMES[c.biome].name,D.biome[c.biome]]);
  if(D.feat&&c.feature&&D.feat[c.feature]) parts.push([FEATN[c.feature],D.feat[c.feature]]);
  parts.push(...aggAdj(null,D,o=>minDistCell(c,o)));
  const own={total:parts.reduce((s,p)=>s+p[1],0),parts};
  const imp=[]; let itotal=0;
  for(const o of buildings){
    if(minDistCell(c,o)<=B[o.type].radius){
      const v=(B[o.type].adj||{})[t];
      if(v){ imp.push([B[o.type].name,v]); itotal+=v; }
    }
  }
  imp.sort((a,b)=>b[1]-a[1]);
  return {own,impact:{total:itotal,parts:imp}};
}

### 📜 4. Менеджер зданий и авто-слияния (`building_manager.gd`)
Реализует ключевую механику: **4 одинаковых здания вплотную → здание II**.

```gdscript
extends Node

var buildings = []

func place_building(type: String, cells: Array) -> Dictionary:
	var b_data = Globals.BUILDINGS[type]
	var new_building = {
		"type": type,
		"cells": cells,
		"is_merged": false,
		"base_type": type,
		"workers": 0,
		"score": b_data.base,
		"house_sp": null
	}
	
	for cell in cells:
		cell.building = new_building
		cell.update_visuals()
		
	buildings.append(new_building)
	return new_building

func try_auto_merge(start_cell) -> Dictionary:
	if not start_cell.building or start_cell.building.is_merged or start_cell.building.cells.size() > 1:
		return {}
		
	var target_type = start_cell.building.type
	var connected = []
	var queue = [start_cell]
	var visited = {start_cell.key: true}
	
	# BFS поиск связных компонентов того же типа
	while not queue.is_empty():
		var current = queue.pop_front()
		connected.append(current)
		
		for dir in Globals.HEX_DIRECTIONS:
			var neighbor = GameManager.get_cell(current.q + dir[0], current.r + dir[1])
			if neighbor and not visited.has(neighbor.key):
				if neighbor.building and not neighbor.building.is_merged and neighbor.building.type == target_type:
					visited[neighbor.key] = true
					queue.append(neighbor)
					
	if connected.size() >= 4:
		var to_merge = connected.slice(0, 4)
		var cells_to_use = to_merge.map(func(c): return c)
		
		# Удаляем старые здания
		for cell in to_merge:
			remove_building(cell.building)
			
		# Создаем "Здание II"
		var merged_data = {
			"type": target_type + "_BIG",
			"cells": cells_to_use,
			"is_merged": true,
			"base_type": target_type,
			"workers": 0,
			"score": int(Globals.BUILDINGS[target_type].base * 5.5),
			"house_sp": null
		}
		
		for cell in cells_to_use:
			cell.building = merged_data
			cell.update_visuals()
			
		buildings.append(merged_data)
		UIManager.spawn_float(cells_to_use[0].position, "Слияние ×4!", "pos")
		return merged_data
		
	return {}

func remove_building(bldg: Dictionary):
	if bldg in buildings:
		buildings.erase(bldg)
	for cell in bldg.cells:
		cell.building = null
		cell.update_visuals()
```

```gdscript
static func hex_to_pixel(q: int, r: int) -> Vector2:
	var x = R * SQRT3 * (q + r * 0.5)
	var y = R * 1.5 * r
	return Vector2(x, y)

static func hex_dist(q1: int, r1: int, q2: int, r2: int) -> int:
	var dq = q1 - q2
	var dr = r1 - r2
	return (abs(dq) + abs(dr) + abs(dq + dr)) / 2

static func get_hex_polygon(center: Vector2, radius: float) -> PackedVector2Array:
	var points = PackedVector2Array()
	for i in range(6):
		var angle_rad = deg_to_rad(60 * i - 30)
		points.append(center + Vector2(cos(angle_rad), sin(angle_rad)) * radius)
	return points
```

### 💡 Как это использовать в Godot:

1. **Создайте файл** `res://globals.gd` и вставьте в него весь код выше.
2. **Добавьте его в Автозагрузку**: `Project` → `Project Settings` → `Autoload` → Добавьте `res://globals.gd` с именем `Globals`.
3. **Доступ из любого скрипта**: Теперь вы можете обращаться к данным из любого места в проекте, например:
   - Получить имя расы: `Globals.SPECIES["Human"]["name"]`
   - Получить цвет биома: `Globals.BIOMES["Rural"]["color"]`
   - Проверить потребности здания: `Globals.BUILDINGS["School"]["prov"]`
   - Вызвать функцию расстояния: `Globals.hex_dist(q1, r1, q2, r2)`

### 📜 2. Менеджер сетки (`res://grid_manager.gd`)
Генерирует карту: ядро 2x2 + 3 ряда, биомы, реки, руины.

```gdscript
extends Node

var cells_dict: Dictionary = {}
signal map_generated

func generate_map():
	# Очистка
	for child in get_parent().get_node("UIManager/RightPanel/GridContainer").get_children():
		child.queue_free()
	cells_dict.clear()
	
	var core = Globals.CORE
	var cell_scene = preload("res://scenes/hex_cell.tscn")
	var grid_container = get_parent().get_node("UIManager/RightPanel/GridContainer")
	
	var seeds = []
	var bag = ["Rural", "Rural", "Urban", "Urban", "Thicket", "Swamp", "Stone", "Arid"]
	
	for b in bag:
		var tries = 0
		var c = null
		while tries < 60:
			var rand_q = randi_range(-6, 6)
			var rand_r = randi_range(-6, 6)
			if _is_valid_core_distance(rand_q, rand_r):
				c = {"q": rand_q, "r": rand_r}
				if not seeds.any(func(s): return Globals.hex_dist(s.q, s.r, c.q, c.r) < 2):
					seeds.append({"b": b, "c": c})
					break
			tries += 1

	for q in range(-6, 7):
		for r in range(-6, 7):
			if not _is_valid_core_distance(q, r): continue
			
			var sorted_seeds = seeds.duplicate()
			sorted_seeds.sort_custom(func(a, b): 
				var da = Globals.hex_dist(a.c.q, a.c.r, q, r) + randf() * 0.6
				var db = Globals.hex_dist(b.c.q, b.c.r, q, r) + randf() * 0.6
				return da < db
			)
			
			var chosen_biome = sorted_seeds[0].b if randf() > 0.12 and sorted_seeds.size() > 1 else sorted_seeds[0].b
			
			var cell = cell_scene.instantiate()
			cell.q = q
			cell.r = r
			cell.ring = _get_ring(q, r)
			cell.biome = chosen_biome
			cell.feature = ""
			
			grid_container.add_child(cell)
			cells_dict[cell.key] = cell
			
			cell.cell_clicked.connect(_on_cell_clicked)
			cell.cell_right_clicked.connect(_on_cell_right_clicked)
			
	_generate_features()
	map_generated.emit()

func _is_valid_core_distance(q: int, r: int) -> bool:
	var min_d = 999
	for c in Globals.CORE:
		var d = Globals.hex_dist(q, r, c[0], c[1])
		if d < min_d: min_d = d
	return min_d <= Globals.RINGS

func _get_ring(q: int, r: int) -> int:
	var min_d = 999
	for c in Globals.CORE:
		var d = Globals.hex_dist(q, r, c[0], c[1])
		if d < min_d: min_d = d
	return min_d

func _generate_features():
	var edge_cells = cells_dict.values().filter(func(c): return c.ring == Globals.RINGS)
	if edge_cells.is_empty(): return
	
	var cur = edge_cells[randi() % edge_cells.size()]
	var dir = Globals.HEX_DIRECTIONS[randi() % 6]
	
	for i in range(8):
		if not cur: break
		cur.feature = "river"
		var want = dir if randf() < 0.7 else Globals.HEX_DIRECTIONS[randi() % 6]
		var nxt = get_cell(cur.q + want[0], cur.r + want[1])
		
		if not nxt or nxt.feature == "river":
			var shuffled_dirs = Globals.HEX_DIRECTIONS.duplicate()
			shuffled_dirs.shuffle()
			for d in shuffled_dirs:
				var t = get_cell(cur.q + d[0], cur.r + d[1])
				if t and t.feature == "":
					nxt = t
					dir = d
					break
		cur = nxt

	var free_cells = cells_dict.values().filter(func(c): return c.feature == "" and c.ring >= 1)
	if not free_cells.is_empty():
		free_cells[randi() % free_cells.size()].feature = "ruins"
		
	var far_cells = cells_dict.values().filter(func(c): 
		return c.feature == "" and c.ring >= 2 and (c.q != free_cells[-1].q or c.r != free_cells[-1].r)
	)
	if not far_cells.is_empty():
		far_cells[randi() % far_cells.size()].feature = "predator"

func get_cell(q: int, r: int):
	return cells_dict.get("%d,%d" % [q, r], null)

func _on_cell_clicked(cell):
	get_parent().get_node("BuildingManager").on_cell_clicked(cell)

func _on_cell_right_clicked(cell):
	get_parent().get_node("BuildingManager").on_cell_right_clicked(cell)
```

### 📜 3. Менеджер зданий и слияний (`res://building_manager.gd`)
Самая сложная часть: размещение, расчет очков (включая соседство), авто-слияние ×4 и ручные рецепты.

```gdscript
extends Node

var buildings: Array = []
var selected_type: String = ""

func on_cell_clicked(cell):
	if selected_type != "":
		if not cell.building.is_empty():
			get_parent().get_node("UIManager").show_status("Клетка занята!")
			return
			
		var new_bldg = place_building(selected_type, [cell])
		var merged = try_auto_merge(cell)
		
		get_parent().get_node("PopulationManager").fill_housing()
		get_parent().get_node("PopulationManager").recalc_all()
		
		if merged:
			get_parent().get_node("UIManager").spawn_float(merged.cells[0].position, "Слияние ×4!", "pos")
		else:
			get_parent().get_node("UIManager").spawn_float(cell.position, Globals.BUILDINGS[selected_type].name, "name")
			
		selected_type = ""
		get_parent().get_node("UIManager").update_palette_selection("")
	else:
		get_parent().get_node("UIManager").show_building_info(cell.building)

func on_cell_right_clicked(cell):
	if not cell.building.is_empty():
		demolish(cell.building)

func place_building(type: String, cells: Array) -> Dictionary:
	var b_data = Globals.BUILDINGS[type]
	var new_building = {
		"type": type,
		"type_deck": b_data.deck,
		"cells": cells,
		"is_merged": b_data.get("merged", false),
		"base_type": type,
		"workers": 0,
		"score": b_data.base,
		"house_sp": null,
		"last_parts": []
	}
	
	for cell in cells:
		cell.building = new_building
		cell.update_visuals()
		
	buildings.append(new_building)
	return new_building

func demolish(bldg: Dictionary):
	get_parent().get_node("PopulationManager").unassign_all_from(bldg)
	buildings.erase(bldg)
	for cell in bldg.cells:
		cell.building = {}
		cell.update_visuals()
		
	get_parent().get_node("PopulationManager").trim_to_housing()
	get_parent().get_node("PopulationManager").recalc_all()
	get_parent().get_node("UIManager").spawn_float(bldg.cells[0].position, "Снесено", "neg")

# --- АВТО-СЛИЯНИЕ ×4 ---
func try_auto_merge(start_cell) -> Dictionary:
	if start_cell.building.is_empty() or start_cell.building.is_merged or start_cell.building.cells.size() > 1:
		return {}
		
	var target_type = start_cell.building.type
	var connected = []
	var queue = [start_cell]
	var visited = {start_cell.key: true}
	
	while not queue.is_empty():
		var current = queue.pop_front()
		connected.append(current)
		
		for dir in Globals.HEX_DIRECTIONS:
			var neighbor = get_parent().get_node("GridManager").get_cell(current.q + dir[0], current.r + dir[1])
			if neighbor and not visited.has(neighbor.key):
				if not neighbor.building.is_empty() and not neighbor.building.is_merged and neighbor.building.type == target_type:
					visited[neighbor.key] = true
					queue.append(neighbor)
					
	if connected.size() >= 4:
		var to_merge = connected.slice(0, 4)
		var cells_to_use = to_merge.duplicate()
		
		# Собираем данные перед удалением
		var carried_workers = []
		var house_sp = to_merge[0].building.house_sp
		for b in to_merge:
			carried_workers.append_array(get_parent().get_node("PopulationManager").get_workers_of(b.building))
			demolish(b.building)
			
		# Создаем "Здание II"
		var base_data = Globals.BUILDINGS[target_type]
		var merged_building = {
			"type": target_type + "_BIG",
			"type_deck": base_data.deck,
			"cells": cells_to_use,
			"is_merged": true,
			"base_type": target_type,
			"workers": 0,
			"score": int(base_data.base * 5.5),
			"house_sp": house_sp,
			"last_parts": []
		}
		
		for cell in cells_to_use:
			cell.building = merged_building
			cell.update_visuals()
			
		buildings.append(merged_building)
		
		# Возвращаем работников
		var max_jobs = base_data.get("jobs", 0) * 4 + 2
		for i in range(min(carried_workers.size(), max_jobs)):
			carried_workers[i]["job"] = merged_building
		merged_building.workers = min(carried_workers.size(), max_jobs)
			
		return merged_building
		
	return {}

# --- РАСЧЕТ ОЧКОВ ---
func recalc_all_scores() -> int:
	var total_score = get_parent().get_node("PopulationManager").rep_score
	for bldg in buildings:
		var score_data = calculate_score(bldg)
		bldg.score = score_data.total
		bldg.last_parts = score_data.parts
		total_score += bldg.score
	return total_score

func calculate_score(bldg: Dictionary) -> Dictionary:
	var b_data = Globals.BUILDINGS[bldg.base_type]
	var anchor = bldg.cells[0]
	var parts = [["База", b_data.base]]
	
	if b_data.has("biome") and b_data.biome.has(anchor.biome):
		parts.append(["Биом: " + Globals.BIOMES[anchor.biome].name, b_data.biome[anchor.biome]])
		
	if b_data.has("feat") and anchor.feature != "" and b_data.feat.has(anchor.feature.capitalize()):
		parts.append([Globals.FEAT_NAMES.get(anchor.feature, anchor.feature), b_data.feat[anchor.feature.capitalize()]])
		
	if bldg.workers > 0 and b_data.has("per_worker"):
		parts.append(["Работники ×%d" % bldg.workers, bldg.workers * b_data.per_worker])
		
	# Соседство
	var adj_parts = get_adjacency_bonus(bldg, b_data)
	for p in adj_parts:
		parts.append(p)
		
	var total = 0
	for p in parts:
		total += p[1]
		
	return {"total": total, "parts": parts}

func get_adjacency_bonus(source_bldg: Dictionary, b_data: Dictionary) -> Array:
	var per_type = {}
	for other in buildings:
		if other == source_bldg: continue
		var dist = min_dist_between(source_bldg, other)
		if dist <= b_data.radius:
			var target_type = other.base_type
			if b_data.has("adj") and b_data.adj.has(target_type):
				var val = b_data.adj[target_type]
				if not per_type.has(target_type):
					per_type[target_type] = {"v": 0, "n": 0}
				per_type[target_type]["v"] += val
				per_type[target_type]["n"] += 1
				
	var result = []
	for t in per_type:
		var name = Globals.BUILDINGS[t].name
		if per_type[t]["n"] > 1:
			name += " ×%d" % per_type[t]["n"]
		result.append([name, per_type[t]["v"]])
		
	result.sort_custom(func(a, b): return a[1] > b[1])
	return result

func min_dist_between(b1: Dictionary, b2: Dictionary) -> int:
	var m = 999
	for c1 in b1.cells:
		for c2 in b2.cells:
			var d = Globals.hex_dist(c1.q, c1.r, c2.q, c2.r)
			if d < m: m = d
	return m
```

### 📜 4. Менеджер населения (`res://population_manager.gd`)
Управляет жителями, жильем, потребностями, моралью (resolve) и сезонами.

```gdscript
extends Node

var villagers: Array = []
var active_species: Array = []
var season: int = 0
var rep_score: int = 0
var storm_active: bool = false
var mourning: int = 0

var housing_stats = {"total": 0, "gen": 0, "frog": 0, "sp_beds": {}}
var needs_stats = {"cap": {}, "dem": {}, "ok": {}}

func _ready():
	init_run()

func init_run():
	villagers.clear()
	active_species = Globals.SPECIES.keys().duplicate()
	active_species.shuffle()
	active_species = active_species.slice(0, 3) # Берем 3 случайные расы
	season = 0
	rep_score = 0
	storm_active = false
	mourning = 0

func next_season():
	season += 1
	storm_active = (season % 3 == 0)
	
	var gained = 0
	for sp in active_species:
		var st = get_species_stat(sp)
		if st.count > 0 and st.avg >= st.demand:
			st.rep += 1
			st.demand += Globals.SPECIES[sp].decadence
			rep_score += 15
			gained += 1
			
	# Проверка морали
	var leavers = villagers.filter(func(v): return v.resolve < 1)
	for v in leavers:
		lose_villager(v)
		
	if leavers.size() > 0:
		get_parent().get_node("UIManager").spawn_float(Vector2.ZERO, "-%d ушло" % leavers.size(), "neg")
		
	# Караван
	if season % 2 == 0:
		var frog_perk = villagers.any(func(v): return not v.job.is_empty() and v.species == "Frog")
		var n = 0
		for i in range(2 + (1 if frog_perk else 0)):
			if push_villager(): n += 1
		if n > 0:
			get_parent().get_node("UIManager").spawn_float(Vector2.ZERO, "+Караван ×%d" % n, "pos")
			
	recalc_all()
	get_parent().get_node("UIManager").show_status("Сезон %d%s" % [season, " · ШТОРМ" if storm_active else ""])

func push_villager() -> bool:
	var cand = active_species.filter(func(sp): return beds_free(sp) > 0)
	if cand.is_empty(): return false
	var sp = cand[randi() % cand.size()]
	villagers.append({"id": randi(), "species": sp, "job": {}, "resolve": 0})
	return true

func lose_villager(v: Dictionary):
	var idx = villagers.find(v)
	if idx >= 0:
		villagers.remove_at(idx)
		if v.species != "Bat": mourning += 1

func fill_housing():
	recompute_housing()
	var added = 0
	for i in range(999): # Guard
		if not push_villager(): break
		added += 1
	return added

func trim_to_housing():
	recompute_housing()
	var removed = 0
	# Сначала убираем лишних жаб
	var frog_cnt = villagers.count(func(v): return v.species == "Frog")
	while frog_cnt > housing_stats.frog:
		var idx = villagers.find_last(func(v): return v.species == "Frog" and v.job.is_empty())
		if idx < 0: idx = villagers.find_last(func(v): return v.species == "Frog")
		if idx < 0: break
		lose_villager(villagers[idx])
		removed += 1
		frog_cnt -= 1
		
	# Потом остальных
	while villagers.size() > housing_stats.total:
		var idx = villagers.find_last(func(v): return v.job.is_empty())
		if idx < 0: idx = 0
		lose_villager(villagers[idx])
		removed += 1
	return removed

func recompute_housing():
	var total = 0; var frog = 0; var sp_beds = {}
	for sp in active_species: sp_beds[sp] = 0
	
	for bldg in get_parent().get_node("BuildingManager").buildings:
		var b_data = Globals.BUILDINGS[bldg.base_type]
		if b_data.has("housing"):
			total += b_data.housing
			if bldg.house_sp:
				sp_beds[bldg.house_sp] += b_data.housing
				if bldg.house_sp == "Frog": frog += b_data.housing
				
	housing_stats = {"total": total, "gen": total - frog, "frog": frog, "sp_beds": sp_beds}

func beds_free(sp: String) -> int:
	if sp == "Frog": return housing_stats.frog - villagers.count(func(v): return v.species == "Frog")
	var non_frog = villagers.count(func(v): return v.species != "Frog")
	return housing_stats.gen - non_frog

func recompute_needs():
	var cap = {"food": 0, "cloth": 0, "Leisure": 0, "Religion": 0, "Education": 0, "Treatment": 0, "Brawling": 0, "Luxury": 0}
	for bldg in get_parent().get_node("BuildingManager").buildings:
		var b_data = Globals.BUILDINGS[bldg.base_type]
		if b_data.has("prov"):
			var ratio = 1.0
			if b_data.has("jobs") and b_data.jobs > 0:
				ratio = min(1.0, float(bldg.workers) / float(b_data.jobs))
			for k in b_data.prov:
				cap[k] += ceil(b_data.prov[k] * ratio)
				
	var dem = {"food": villagers.size(), "cloth": villagers.size()}
	for v in villagers:
		for s in Globals.SPECIES[v.species].services:
			dem[s] = dem.get(s, 0) + 1
			
	var ok = {}
	for k in cap:
		ok[k] = (dem.get(k, 0) == 0) or (cap[k] >= dem.get(k, 0))
		
	needs_stats = {"cap": cap, "dem": dem, "ok": ok}

func recalc_resolve():
	recompute_housing()
	recompute_needs()
	
	var human_perk = villagers.any(func(v): return not v.job.is_empty() and v.species == "Human")
	var lizard_perk = villagers.any(func(v): return not v.job.is_empty() and v.species == "Lizard")
	var bat_bonus = min(6, floor(mourning / 2.0))
	
	for v in villagers:
		var sp_data = Globals.SPECIES[v.species]
		var r = sp_data.base
		r += 5 if needs_stats.ok.food else -10
		r += 3 if needs_stats.ok.cloth else -8
		
		for s in sp_data.services:
			r += 3 if needs_stats.ok.get(s, false) else -6
			
		if v.species == "Frog":
			var ratio = housing_stats.frog > 0 ? min(1.0, float(housing_stats.frog) / float(villagers.count(func(x): return x.species == "Frog"))) : 1.0
			r += 8 if ratio >= 1.0 else (ceil(8 * ratio) if ratio > 0 else -12)
		else:
			var non_frog = villagers.count(func(x): return x.species != "Frog")
			var shelter_ok = housing_stats.gen >= non_frog
			r += 4 if shelter_ok else -10
			var cap = v.species == "Frog" ? housing_stats.frog : housing_stats.sp_beds.get(v.species, 0)
			var cnt = villagers.count(func(x): return x.species == v.species)
			var ratio = cnt > 0 ? min(1.0, float(cap) / float(cnt)) : 1.0
			r += 8 if ratio >= 1.0 else (ceil(8 * ratio) if ratio > 0 else -6)
			
		if storm_active:
			var pen = Globals.STORM_PEN[sp_data.res] * (0.75 if human_perk else 1.0)
			r -= round(pen * 10.0) / 10.0
			
		if lizard_perk: r += 1
		if v.species == "Bat": r += bat_bonus
		
		if not v.job.is_empty():
			var j_data = Globals.BUILDINGS[v.job.base_type]
			if j_data.get("spec") == sp_data.comfort: r += 5
			
		v.resolve = r

func recalc_all():
	for bldg in get_parent().get_node("BuildingManager").buildings:
		bldg.workers = villagers.count(func(v): return v.job == bldg)
		
	recalc_resolve()
	
	var total = recalc_all_scores() # Вызываем из BuildingManager
	get_parent().get_node("UIManager").update_ui(total, villagers.size(), rep_score)

func get_species_stat(sp: String) -> Dictionary:
	var vs = villagers.filter(func(v): return v.species == sp)
	var cnt = vs.size()
	var avg = 0.0
	if cnt > 0:
		var sum = 0
		for v in vs: sum += v.resolve
		avg = sum / float(cnt)
	return {"count": cnt, "avg": avg, "demand": Globals.SPECIES[sp].demand, "rep": 0} # rep упрощен для примера

func get_workers_of(bldg: Dictionary) -> Array:
	return villagers.filter(func(v): return v.job == bldg)

func unassign_all_from(bldg: Dictionary):
	for v in villagers:
		if v.job == bldg: v.job = {}
```

### 🚀 Как запустить и проверить:

1. Убедитесь, что `globals.gd` добавлен в **Autoload**.
2. Соберите сцену `Main.tscn` согласно схеме выше.
3. Назначьте скрипты соответствующим узлам.
4. Запустите проект (`F5`).
5. **Проверка механик:**
   - Клик по кнопке в левой панели выбирает здание.
   - Клик по пустой клетке ставит здание.
   - Поставьте 4 одинаковых здания (например, "Школа") вплотную друг к другу. Четвертое автоматически превратится в "Школа_BIG" с увеличенными характеристиками.
   - Клик по зданию показывает карточку с расчетом очков (база + биом + соседи).
   - Нажмите кнопку "Сезон +" (добавьте её в UI), чтобы проверить начисление славы и появление жителей.

Этот код полностью воспроизводит логику вашего HTML-прототипа, но использует производительные и идиоматичные для Godot 4 подходы (словари для данных, `Tween` для анимаций, сигналы для взаимодействия). Если нужно добавить визуальные спрайты вместо текстовых заглушек, дайте знать!
